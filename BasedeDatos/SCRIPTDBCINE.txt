CREATE DATABASE cine;
USE cine;

CREATE TABLE rol (
	id_rol INT IDENTITY(1,1) PRIMARY KEY,
	nombre_rol NVARCHAR(44) NOT NULL UNIQUE
);

CREATE TABLE usuario (
    id_usuario INT IDENTITY(1,1) PRIMARY KEY,
    nombre NVARCHAR(50) NOT NULL,
    correo NVARCHAR(100) NOT NULL UNIQUE,
    contrasena NVARCHAR(255) NOT NULL, 
    imagen NVARCHAR(255),
    id_rol INT NOT NULL FOREIGN KEY REFERENCES rol(id_rol)
);


CREATE TABLE pelicula (
    id_pelicula INT PRIMARY KEY IDENTITY(1,1),
    titulo NVARCHAR(150) NOT NULL,
    descripcion NVARCHAR(MAX),
    duracion INT,
    genero NVARCHAR(50),
    fecha_estreno DATE,
    director NVARCHAR(100),
    imagen NVARCHAR(255)
);

CREATE TABLE sala (
    id_sala INT PRIMARY KEY IDENTITY(1,1),
    nombre NVARCHAR(50) NOT NULL,
    capacidad INT NOT NULL,
    filas INT NOT NULL,
    columnas INT NOT NULL,
    tipo NVARCHAR(30),
    estado BIT NOT NULL DEFAULT 1
);

CREATE TABLE funcion (
    id_funcion INT PRIMARY KEY IDENTITY(1,1),	
    descripcion NVARCHAR(50),
    id_pelicula INT NOT NULL FOREIGN KEY REFERENCES pelicula(id_pelicula),
    id_sala INT NOT NULL FOREIGN KEY REFERENCES sala(id_sala),
    fecha DATE NOT NULL,
    hora NVARCHAR(10) NOT NULL
);

CREATE TABLE asiento (
    id_asiento INT PRIMARY KEY IDENTITY(1,1),
    id_funcion INT NOT NULL FOREIGN KEY REFERENCES funcion(id_funcion),
    codigo NVARCHAR(10) NOT NULL,
    disponible BIT NOT NULL DEFAULT 1
);

CREATE TABLE reserva (
    id_reserva INT PRIMARY KEY IDENTITY(1,1),
    id_usuario INT NOT NULL FOREIGN KEY REFERENCES usuario(id_usuario),
    id_funcion INT NOT NULL FOREIGN KEY REFERENCES funcion(id_funcion),
    fecha_reserva DATETIME NOT NULL DEFAULT GETDATE()
);

CREATE TABLE reserva_asiento (
    id_reserva INT NOT NULL FOREIGN KEY REFERENCES reserva(id_reserva),
    id_asiento INT NOT NULL FOREIGN KEY REFERENCES asiento(id_asiento),
    PRIMARY KEY (id_reserva, id_asiento)
);
